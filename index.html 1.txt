<script>

let startTime = 0;
let elapsed = 0;
let interval;
let running = false;
let pauseCount = 0;
let laps = [];

const display = document.getElementById("display");

function formatTime(ms){

let milliseconds = ms % 1000;

let seconds = Math.floor(ms/1000)%60;

let minutes = Math.floor(ms/60000)%60;

let hours = Math.floor(ms/3600000);

return (
String(hours).padStart(2,'0') + ":" +
String(minutes).padStart(2,'0') + ":" +
String(seconds).padStart(2,'0') + "." +
String(milliseconds).padStart(3,'0')
);
}

function startTimer(){

if(running) return;

running = true;

startTime = Date.now() - elapsed;

interval = setInterval(()=>{

elapsed = Date.now()-startTime;

display.textContent = formatTime(elapsed);

updateDashboard();

checkMilestones();

},10);
}

function pauseTimer(){

if(!running) return;

running = false;

pauseCount++;

clearInterval(interval);
}

function resetTimer(){

clearInterval(interval);

saveSession();

running = false;
elapsed = 0;
laps = [];

display.textContent="00:00:00.000";
document.getElementById("laps").innerHTML="";

updateDashboard();
}

function saveLap(){

if(elapsed===0) return;

laps.push(formatTime(elapsed));

let li=document.createElement("li");

li.textContent="Lap "+laps.length+" : "+formatTime(elapsed);

document.getElementById("laps").prepend(li);
}

function updateDashboard(){

document.getElementById("totalTime").innerText=
formatTime(elapsed).split(".")[0];

let score=Math.max(0,100-(pauseCount*5));

document.getElementById("score").innerText=score+"%";

document.getElementById("currentActivity").innerText=
document.getElementById("activity").value;
}

let notified = [];

function checkMilestones(){

const mins = Math.floor(elapsed/60000);

if(mins===10 && !notified.includes(10)){
alert("?? Great Job! 10 Minutes Focused");
notified.push(10);
}

if(mins===30 && !notified.includes(30)){
alert("?? Amazing! 30 Minutes Focused");
notified.push(30);
}

if(mins===60 && !notified.includes(60)){
alert("?? One Hour Focus Session Completed");
notified.push(60);
}
}

function saveSession(){

if(elapsed===0) return;

const history=
JSON.parse(localStorage.getItem("focusHistory")) || [];

history.push({
activity:document.getElementById("activity").value,
time:formatTime(elapsed),
score:Math.max(0,100-(pauseCount*5)),
date:new Date().toLocaleString(),
laps:laps.length
});

localStorage.setItem(
"focusHistory",
JSON.stringify(history)
);

loadHistory();
}

function loadHistory(){

const history=
JSON.parse(localStorage.getItem("focusHistory")) || [];

let html="";

history.reverse().forEach(item=>{

html+=`
<div class="history-item">
<strong>${item.activity}</strong><br>
Time: ${item.time}<br>
Score: ${item.score}%<br>
Laps: ${item.laps}<br>
Date: ${item.date}
</div>
`;
});

document.getElementById("history").innerHTML=html;
}

function exportCSV(){

const history=
JSON.parse(localStorage.getItem("focusHistory")) || [];

let csv=
"Activity,Time,Score,Laps,Date\n";

history.forEach(item=>{

csv += `${item.activity},${item.time},${item.score},${item.laps},${item.date}\n`;

});

const blob = new Blob([csv], {type:'text/csv'});

const url = URL.createObjectURL(blob);

const a=document.createElement('a');

a.href=url;

a.download="FocusTrack_History.csv";

a.click();
}

function toggleTheme(){
document.body.classList.toggle("light");
}

loadHistory();
updateDashboard();
